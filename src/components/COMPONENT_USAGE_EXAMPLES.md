# Component Usage Examples

## PickSelector Component

### Basic Usage in GameCard.tsx

```tsx
import PickSelector from './PickSelector';
import { submitPick } from '../utils/apiClient';

// Inside your GameCard component:
const handlePickSubmit = async (teamId: string) => {
  try {
    await submitPick(game.id, teamId);
    // Optionally refetch game data or update local state
    onPickUpdated?.();
  } catch (error) {
    // Error is handled internally by PickSelector
    throw error;
  }
};

return (
  <div className="game-card">
    {/* Other game info */}

    <PickSelector
      game={{
        id: game.id,
        homeTeam: game.homeTeam,
        awayTeam: game.awayTeam,
        isLocked: game.isLocked,
      }}
      currentPick={game.userPick}
      onPickSubmit={handlePickSubmit}
      disabled={!isAuthenticated}
    />
  </div>
);
```

### State Management

```tsx
// Example with optimistic updates and local state
const [localPick, setLocalPick] = useState(game.userPick);

const handlePickSubmit = async (teamId: string) => {
  const previousPick = localPick;

  // Optimistic update
  setLocalPick({ teamId, autoGenerated: false });

  try {
    await submitPick(game.id, teamId);
  } catch (error) {
    // Rollback on error
    setLocalPick(previousPick);
    throw error;
  }
};

<PickSelector
  game={game}
  currentPick={localPick}
  onPickSubmit={handlePickSubmit}
/>
```

### Complete GameCard Integration Example

```tsx
import React from 'react';
import PickSelector from './PickSelector';
import GameResult from './GameResult';
import { submitPick } from '../utils/apiClient';

interface GameCardProps {
  game: {
    id: string;
    homeTeam: { id: string; name: string; abbreviation: string; logoUrl?: string };
    awayTeam: { id: string; name: string; abbreviation: string; logoUrl?: string };
    startTime: string;
    status: 'scheduled' | 'in_progress' | 'final';
    isLocked: boolean;
    homeScore?: number;
    awayScore?: number;
    homeSpread?: number;
    overUnder?: number;
    winnerTeamId?: string;
    userPick?: {
      teamId: string;
      autoGenerated?: boolean;
      isCorrect?: boolean;
    };
  };
  onPickUpdated?: () => void;
}

export default function GameCard({ game, onPickUpdated }: GameCardProps) {
  const handlePickSubmit = async (teamId: string) => {
    try {
      await submitPick(game.id, teamId);
      onPickUpdated?.();
    } catch (error) {
      throw error; // PickSelector handles error display
    }
  };

  const showResult = game.status === 'final' || game.status === 'in_progress';

  return (
    <div className="rounded-lg border border-gray-200 bg-white p-4 shadow-sm">
      {/* Game Header */}
      <div className="mb-4 flex items-center justify-between">
        <div className="text-sm font-medium text-gray-600">
          {new Date(game.startTime).toLocaleDateString('en-US', {
            weekday: 'short',
            month: 'short',
            day: 'numeric',
            hour: 'numeric',
            minute: '2-digit',
          })}
        </div>
        <div className="text-xs text-gray-500">
          {game.status === 'scheduled' && 'Upcoming'}
          {game.status === 'in_progress' && 'Live'}
          {game.status === 'final' && 'Final'}
        </div>
      </div>

      {/* Pick Selector or Game Result */}
      {showResult ? (
        <GameResult
          game={{
            id: game.id,
            homeTeam: game.homeTeam,
            awayTeam: game.awayTeam,
            homeScore: game.homeScore ?? 0,
            awayScore: game.awayScore ?? 0,
            homeSpread: game.homeSpread,
            overUnder: game.overUnder,
            status: game.status,
            winnerTeamId: game.winnerTeamId,
          }}
          userPick={game.userPick}
        />
      ) : (
        <PickSelector
          game={{
            id: game.id,
            homeTeam: game.homeTeam,
            awayTeam: game.awayTeam,
            isLocked: game.isLocked,
          }}
          currentPick={game.userPick}
          onPickSubmit={handlePickSubmit}
        />
      )}
    </div>
  );
}
```

describe('PickSelector', () => {

    );

## Migration from Inline Implementation

If you have existing inline pick selection in GameCard.tsx:

1. Import the new components:
   ```tsx
   import PickSelector from './PickSelector';
   import GameResult from './GameResult';
   ```

2. Replace inline pick buttons with `<PickSelector />`

3. Replace inline score display with `<GameResult />`

4. Remove duplicate state management code (PickSelector handles it internally)

5. Update your API client wrapper to match the `onPickSubmit` signature

## Troubleshooting

**Issue:** Picks not saving
- Check that `onPickSubmit` throws errors (not swallowing them)
- Verify API client is properly configured
- Check network tab for 401/403 authentication errors

**Issue:** Components not rendering
- Verify all required props are provided
- Check browser console for error boundary messages
- Ensure team data includes all required fields

**Issue:** Styling conflicts
- Check that Tailwind CSS is properly configured
- Verify no global styles are overriding component styles
- Use browser dev tools to inspect computed styles
